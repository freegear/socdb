cramfsck :
	this file is the utility for CRAMFS file system
	this extracts CRAMFS image to temporary directory
	only for linux OS, it doesn't work on Windows

	cramfsck  [dir. name]  [cramfs image]


mkcramfs :
	make CRAMFS file system image
	only for linux OS, it doesn't work on Windows

	cramfs  [dir. name]  [the name that is made]


root-mba2440.cramfs :
	CRAMFS file system image for mba2440
